<script>
    // Parámetros de la Calculadora (Valores basados en promedios de consumo y factores de emisión)
    const CONSUMPTION_RATES = {
        carro: 2.0,       // Gal/h (Gasolina - SUV promedio)
        bus: 0.25,        // Gal/h (Diésel - 1/20 de la emisión total del bus, asumiendo 20 pasajeros)
        moto: 0.75,       // Gal/h (Gasolina - Moto promedio)
        camioneta: 3.0,   // Gal/h (Diésel/Gasolina - Camioneta promedio)
        lancha: 4.9,      // Gal/h (Diésel - Lancha 100 CV)
        avion: 12.0       // Gal/h (Turbosina/Diésel - 1/300 de la emisión total del avión, asumiendo 300 pasajeros)
    };

    const EMISSION_FACTORS = {
        carro: 8.89,      // Kg CO2/Gal (Gasolina)
        bus: 10.18,       // Kg CO2/Gal (Diésel)
        moto: 8.89,
        camioneta: 10.18,
        lancha: 10.18,
        avion: 10.18      // Usamos Diésel como proxy para Turbosina
    };

    // Umbrales para la Huella de Carbono MENSUAL (Kg CO2/Mes)
    const UMBRAL_AZUL = 1;        // Prácticamente cero (solo para distinción visual)
    const UMBRAL_VERDE = 400;     // Bajo: 1 - 400 kg CO2/mes
    const UMBRAL_AMARILLO = 1500; // Medio: 400 - 1500 kg CO2/mes
    // Alto: > 1500 kg CO2/mes

    /**
     * Obtiene el valor numérico (días a la semana) seleccionado del radio button.
     * @param {string} name - Nombre del grupo de radio buttons (e.g., 'dias_carro').
     * @returns {number} El número de días a la semana (0, 2, 4, 6, o 7).
     */
    function getRadioValue(name) {
        const selected = document.querySelector(`input[name="${name}"]:checked`);
        // Usamos Math.max para asegurar que si no se selecciona nada (o falla), sea 0.
        return selected ? Math.max(0, parseInt(selected.value)) : 0;
    }

    /**
     * Calcula la huella de carbono mensual y la clasifica con el semáforo.
     */
    function calcularHuellaMensual() {
        let huella_co2_total = 0;
        const calculos_individuales = [];

        // Definición de vehículos con inputs de horas diarias y radio buttons de días
        const vehiculos_diarios = [
            { tipo: 'carro', label: 'Carro/SUV' },
            { tipo: 'bus', label: 'Bus' },
            { tipo: 'moto', label: 'Moto' },
            { tipo: 'camioneta', label: 'Camioneta' }
        ];

        // 1. Cálculo de Vehículos de Uso Diario/Semanal
        vehiculos_diarios.forEach(({ tipo, label }) => {
            const horas_dia = parseFloat(document.getElementById(`horas_${tipo}`).value) || 0;
            const dias_semana = getRadioValue(`dias_${tipo}`);

            if (horas_dia > 0 && dias_semana > 0) {
                const consumo_h = CONSUMPTION_RATES[tipo];
                const factor_co2 = EMISSION_FACTORS[tipo];

                // Consumo Semanal (Gal) = Consumo_h * Horas_dia * Días_semana
                const consumo_semanal = consumo_h * horas_dia * dias_semana;

                // Emisión Mensual (Kg CO2/Mes) = Consumo_semanal * 4 semanas/mes * Factor_CO2
                const emision_mensual = consumo_semanal * 4 * factor_co2;
                
                huella_co2_total += emision_mensual;
                calculos_individuales.push(`${label}: ${emision_mensual.toFixed(2)} kg CO₂`);
            }
        });

        // 2. Cálculo de Avión (Uso Mensual Directo)
        // Se asume que las 'horas_avion' son totales en el mes.
        const horas_avion = parseFloat(document.getElementById('horas_avion').value) || 0;
        if (horas_avion > 0) {
            const consumo_h = CONSUMPTION_RATES.avion;
            const factor_co2 = EMISSION_FACTORS.avion;

            // Emisión Mensual (Kg CO2/Mes) = Consumo_h * Horas_mes * Factor_CO2
            const emision_mensual_avion = horas_avion * consumo_h * factor_co2;
            huella_co2_total += emision_mensual_avion;
            calculos_individuales.push(`Avión: ${emision_mensual_avion.toFixed(2)} kg CO₂`);
        }
        
        // 3. Cálculo de Lancha (Uso Mensual Directo)
        // Se asume que las 'horas_lancha' son totales en el mes.
        const horas_lancha = parseFloat(document.getElementById('horas_lancha').value) || 0;
        if (horas_lancha > 0) {
            const consumo_h = CONSUMPTION_RATES.lancha;
            const factor_co2 = EMISSION_FACTORS.lancha;

            // Emisión Mensual (Kg CO2/Mes) = Consumo_h * Horas_mes * Factor_CO2
            const emision_mensual_lancha = horas_lancha * consumo_h * factor_co2;
            huella_co2_total += emision_mensual_lancha;
            calculos_individuales.push(`Lancha: ${emision_mensual_lancha.toFixed(2)} kg CO₂`);
        }

        // 4. Clasificación del Semáforo
        let colorSemaforo, clasificacion, claseTexto;
        const dias_cero = getRadioValue('dias_cero');

        if (huella_co2_total < UMBRAL_AZUL) { // Casi Cero o Cero
            colorSemaforo = 'color-azul';
            clasificacion = 'CERO/BAJAS EMISIONES';
            claseTexto = 'texto-azul';
        } else if (huella_co2_total < UMBRAL_VERDE) {
            colorSemaforo = 'color-verde';
            clasificacion = 'HUELLA BAJA (No muy contaminante)';
            claseTexto = 'texto-verde';
        } else if (huella_co2_total >= UMBRAL_VERDE && huella_co2_total <= UMBRAL_AMARILLO) {
            colorSemaforo = 'color-amarillo';
            clasificacion = 'HUELLA MEDIA (Contaminante Medio)';
            claseTexto = 'texto-amarillo';
        } else {
            colorSemaforo = 'color-rojo-oscuro';
            clasificacion = 'HUELLA ALTA (Muy Contaminante)';
            claseTexto = 'texto-rojo-oscuro';
        }

        // 5. Mostrar Resultados
        const resultadoDiv = document.getElementById('resultado');
        const detalles = calculos_individuales.length > 0 ? `
            <h4>Desglose Mensual de Emisiones por Transporte:</h4>
            <ul>
                ${calculos_individuales.map(item => `<li>${item}</li>`).join('')}
            </ul>
        ` : 'No se registraron emisiones de transporte por combustión, ¡Excelente!';

        resultadoDiv.innerHTML = `
            <h2>Resultado de su Huella Mensual</h2>
            <p>Uso de Cero Emisiones (Bici/Caminata/Eléctrico): <strong>${dias_cero}</strong> días a la semana.</p>
            <hr>
            <h3>Huella de CO₂ Total Estimada:</h3>
            <h1><strong>${huella_co2_total.toFixed(2)} kg CO₂/mes</strong></h1>
            
            <p>Clasificación del Semáforo:
                <span class="${claseTexto}"><strong>${clasificacion}</strong></span>
                <span class="semaforo ${colorSemaforo}"></span>
            </p>
            ${detalles}
            <p class="nota">⚠️ *Nota: Los cálculos de Bus y Avión asumen que la emisión se divide entre un número promedio de pasajeros. La calificación del semáforo se basa en umbrales de consumo típicos para un solo individuo y son estimados.</p>
        `;
    }

    // Inicializar el resultado al cargar la página
    document.addEventListener('DOMContentLoaded', () => {
        document.getElementById('resultado').innerHTML = '<h2>¡Listo para calcular!</h2><p>Por favor, ingrese sus hábitos de transporte y presione "Calcular Huella Mensual".</p>';
    });
</script>
